import CodeBlock from '@theme/CodeBlock'

Examples

| Class                                                         | Description                                                                                                                                                                      |
| ------------------------------------------------------------- | -------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| <CodeBlock language="css">.md:is-col-6</CodeBlock>            | Sets the component's width to a `6/12` ratio for screens larger than `md`.                                                                                          |
| <CodeBlock language="css">.is-col-12 .md:is-col-4</CodeBlock> | Sets the component's width to a `12/12` ratio (i.e., `100%`) for `mobile` screens and a `4/12` ratio when the screen is larger than `md`. |
