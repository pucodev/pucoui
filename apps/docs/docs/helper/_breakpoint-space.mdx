import CodeBlock from '@theme/CodeBlock'

Examples

| Class                                               | Description                                                                                                                                        |
| --------------------------------------------------- | -------------------------------------------------------------------------------------------------------------------------------------------------- |
| <CodeBlock language="css">.sm:pt-2</CodeBlock>      | Adds a `padding-top` of `space 2` when the screen is larger than `sm`.                                                                            |
| <CodeBlock language="css">pb-3 .lg:pb-2</CodeBlock> | Adds a `padding-bottom` of `space 3` for `mobile` screens and sets the `padding-bottom` to `space 2` when the screen is larger than `lg`. |
