import CodeBlock from '@theme/CodeBlock'

Examples

| Class                                                       | Description                                                                                                  |
| ----------------------------------------------------------- | ------------------------------------------------------------------------------------------------------------ |
| <CodeBlock language="css">.md:is-hidden</CodeBlock>         | Hides the element (`display: none`) for screens larger than `md`.                                           |
| <CodeBlock language="css">is-hidden .sm:is-flex</CodeBlock> | Hides the element for `mobile` screens and makes it a flex container (`display: flex`) for screens larger than `md`. |
